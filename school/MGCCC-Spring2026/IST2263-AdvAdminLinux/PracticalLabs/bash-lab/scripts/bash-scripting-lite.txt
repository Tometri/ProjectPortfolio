    echo "Error: '$DIR' is not a direct>

for ITEM in "$DIR"/*; do
    # Skip if glob didn't match anything
    [ ! -e "$ITEM" ] && continue

    echo "Item: $ITEM"

    if [ -r "$ITEM" ]; then
        echo "  readable: yes"
    else
        echo "  readable: no"
    fi

    if [ -w "$ITEM" ]; then
        echo "  writable: yes"
    else
        echo "  writable: no"
    fi

    if [ -x "$ITEM" ]; then
        echo "  executable: yes"
    else
        echo "  executable: no"
    fi

    echo
done
